import 'package:flutter/foundation.dart';
import 'package:google_sign_in/google_sign_in.dart';

/// Thin wrapper around the google_sign_in **v7** API.
///
/// Flow:
///  1. [attemptLightweightAuthentication] – restores a previous session with
///     minimal / no UI (equivalent to the old `signInSilently`).
///  2. If that returns null, fall through to the full interactive [authenticate].
///
/// The Web Client ID is passed as [serverClientId] on Android / iOS so that
/// the backend can verify the resulting ID token.
abstract final class GoogleAuthService {
  static const _webClientId =
      '1042945824102-jtp8tdh929gdb89n5aoji4qi6tdro811.apps.googleusercontent.com';

  static bool _initialized = false;

  static Future<void> _ensureInitialized() async {
    if (_initialized) return;
    await GoogleSignIn.instance.initialize(
      // On web the clientId must be set; on Android/iOS the serverClientId
      // is what allows the backend to receive a verifiable ID token.
      clientId: kIsWeb ? _webClientId : null,
      serverClientId: kIsWeb ? null : _webClientId,
    );
    _initialized = true;
  }

  /// Signs the user in with Google and returns the ID token string, or `null`
  /// if the user explicitly cancelled the picker.
  ///
  /// Throws a [GoogleAuthException] on configuration or network errors.
  static Future<String?> getIdToken() async {
    await _ensureInitialized();

    GoogleSignInAccount? account;

    // ── 1. Try lightweight / silent restore first ───────────────────────────
    // attemptLightweightAuthentication returns a Future? (nullable future).
    // If the platform doesn't support it, the method itself returns null.
    final silentFuture = GoogleSignIn.instance
        .attemptLightweightAuthentication();
    if (silentFuture != null) {
      try {
        account = await silentFuture;
      } catch (_) {
        // Silent path failing is normal – fall through to interactive.
      }
    }

    // ── 2. Interactive sign-in when silent fails or is unsupported ──────────
    account ??= await GoogleSignIn.instance.authenticate();

    final idToken = account.authentication.idToken;
    if (idToken == null) {
      throw const GoogleAuthException(
        'Google did not return an ID token. '
        'Ensure the Web Client ID is set as serverClientId and that '
        'google-services.json is present in android/app/.',
      );
    }
    return idToken;
  }

  /// Signs the user out of Google (clears cached credentials).
  static Future<void> signOut() async {
    if (!_initialized) return;
    try {
      await GoogleSignIn.instance.signOut();
    } catch (_) {}
  }
}

/// Thrown by [GoogleAuthService] on configuration or network errors.
class GoogleAuthException implements Exception {
  const GoogleAuthException(this.message);
  final String message;

  @override
  String toString() => message;
}

      Api.post('/auth/resend-otp', {'phone': phone});

  /// On success stores the token and returns the result; `profile_complete`
  /// in the data tells the caller whether step 3 is needed.
  static Future<ApiResult> verifyOtp(String phone, String otp) async {
    final result = await Api.post('/auth/verify-otp', {
      'phone': phone,
      'otp': otp,

    if (result.ok) {

      Api.token = t;
      if (t != null) {
        await TokenStore.write(t);
      }
    }
    return result;
  }

      email: '${data['email'] ?? ''}',
      phone: '${data['phone'] ?? ''}',
      city: cityJson is Map

      age: data['age'] is int ? data['age'] as int : null,
      vipTier: '${vip['tier'] ?? data['vip_tier'] ?? 'silver'}',

  }

  bool get _validPhone {
    return RegExp(r'^1[0125]\d{8}$').hasMatch(_normalizedPhoneDigits);
  }

  String get _normalizedPhoneDigits {
    final digits = _phone.text.replaceAll(RegExp(r'\D'), '');
    return digits.startsWith('0') ? digits.substring(1) : digits;
  }

  Future<void> _continue() async {
    if (!_validPhone) {
      showAppSnack(context, tr(context, 'auth.invalid_phone'), error: true);
      return;
    }
    if (!_agreed) {
      showAppSnack(context, tr(context, 'auth.accept_terms'), error: true);
      return;
    }
    final digits = _normalizedPhoneDigits;
    setState(() => _sending = true);
    // Real OTP via backend (SMS Misr); when the backend is unreachable the
    // flow continues in mock mode where any 4-digit code verifies.
    final result = await BackendGateway.sendOtp('+20$digits');
    if (!mounted) return;
    setState(() => _sending = false);
    if (!result.ok && !result.offline) {
      showAppSnack(context, result.message ?? 'Error', error: true);
      return;
    }
    Navigator.of(
      context,
    ).push(MaterialPageRoute(builder: (_) => OtpScreen(phone: '+20 $digits')));
  }

  Future<void> _google() async {
    if (!_agreed) {
      showAppSnack(context, tr(context, 'auth.accept_terms'), error: true);
      return;
    }
    setState(() => _sending = true);
    try {
      final token = await GoogleAuthService.getIdToken();
      if (token == null) return;
      final result = await BackendGateway.googleLogin(token);
      if (!mounted) return;
      if (!result.ok) {
        throw StateError(result.message ?? 'Google sign-in failed.');
      }
      final profileResult = await BackendGateway.fetchProfile();
      final profile = BackendGateway.profileFromJson(profileResult.data);
      if (!mounted || profile == null) return;
      Navigator.of(context).push(
        MaterialPageRoute(
          builder: (_) => CompleteProfileScreen(
            initialName: profile.fullName,
            initialEmail: profile.email,
            emailVerified: true,
          ),
        ),
      );
    } catch (e) {
      if (mounted) showAppSnack(context, e.toString(), error: true);
